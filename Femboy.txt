-- Helper to check if a feature is active based on Mode
local function isFeatureActive(enabled, key, mode, toggleState)
    if not enabled then return false end
    if mode == "Toggle" then
        return toggleState
    else
        return UserInputService:IsKeyDown(key)
    end
end -- [cite: 212]

local function isAlive(plr)
    if not (plr and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character:FindFirstChild("HumanoidRootPart")) then
        return false
    end
    if plr.Character.Humanoid.Health <= 0 then return false end
    if plr.Character.HumanoidRootPart.Position.Y < -5 then return false end
 
    return true -- [cite: 223]
end

local function isVisible(part, character)
    if not state.wallCheck then return true end
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, Camera, character}
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(Camera.CFrame.Position, (part.Position - Camera.CFrame.Position), rayParams)
    return result == nil -- [cite: 223]
end

local function getClosestToMouse()
    local target, shortestDist = nil, state.fovRadius
    local mouseLoc = UserInputService:GetMouseLocation()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and isAlive(plr) and (not state.enemyOnly or not sameTeam(plr)) then -- [cite: 223, 224]
            local part = plr.Character:FindFirstChild(state.aimPart)
            if part then
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - mouseLoc).Magnitude
    
                    if dist < shortestDist and isVisible(part, plr.Character) then -- [cite: 224, 225]
                        shortestDist = dist;
                        target = plr -- [cite: 226]
                    end
                end
            end
        end
    end
    return target -- [cite: 226]
end
