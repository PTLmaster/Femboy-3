-- Helper to check if a feature is active based on Mode
local function isFeatureActive(enabled, key, mode, toggleState)
    if not enabled then return false end
    if mode == "Toggle" then
        return toggleState
    else
        return UserInputService:IsKeyDown(key)
    end
end

-- =========================
-- Optimized Weapon Mods Loop
-- =========================
task.spawn(function()
    while true do
        if state.active and state.isLoggedIn then
            local weapons = ReplicatedStorage:FindFirstChild("Weapons")
            if weapons then
                local isRapidActive = isFeatureActive(state.rapidFireEnabled, state.rapidFireKey, state.rapidMode, state.rapidToggled)
                
                for _, v in pairs(weapons:GetDescendants()) do
                    if v:IsA("ValueBase") then
                        if not originalStats[v] then
                            if v.Name == "Auto" or v.Name == "FireRate" or v.Name == "RecoilControl" or v.Name == "MaxSpread" then
                                originalStats[v] = v.Value
                            end
                        end

                        if v.Name == "RecoilControl" then
                            v.Value = state.recoil and 0 or (originalStats[v] or v.Value)
                        elseif v.Name == "MaxSpread" then
                            v.Value = state.spread and 0 or (originalStats[v] or v.Value)
                        elseif v.Name == "Auto" then
                            v.Value = isRapidActive and true or (originalStats[v] or v.Value)
                        elseif v.Name == "FireRate" then
                            v.Value = isRapidActive and 0.02 or (originalStats[v] or v.Value)
                        end
                    end
                end
            end
        end
        task.wait(0.5)
    end
end)

-- Handle Toggle Logic for Inputs
UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    
    -- Rapid Fire Toggle
    if input.KeyCode == state.rapidFireKey and state.rapidMode == "Toggle" then
        state.rapidToggled = not state.rapidToggled
    end
    -- Speed Toggle
    if input.KeyCode == state.speedKey and state.speedMode == "Toggle" then
        state.speedToggled = not state.speedToggled
    end
    -- Fly Toggle
    if input.KeyCode == state.flyKey and state.flyMode == "Toggle" then
        state.flyToggled = not state.flyToggled
    end
end)

local function cleanupPlayer(plr)
    if bonesData[plr] then
        if bonesData[plr].limbs then
            for _, line in pairs(bonesData[plr].limbs) do
                line.Visible = false
                line:Remove()
            end
        end
        bonesData[plr] = nil
    end

    local char = plr.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            if root:FindFirstChild("BnuuyBox") then root.BnuuyBox:Destroy() end
            if root:FindFirstChild("BnuuyName") then root.BnuuyName:Destroy() end
            if root:FindFirstChild("BnuuyDistance") then root.BnuuyDistance:Destroy() end
        end
        if char:FindFirstChild("BnuuyChams") then char.BnuuyChams:Destroy() end
    end
    deadPlayers[plr] = nil
end

local function sameTeam(plr)
    return plr.Team == LocalPlayer.Team or plr.TeamColor == LocalPlayer.TeamColor 
end

local function isAlive(plr)
    if not (plr and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character:FindFirstChild("HumanoidRootPart")) then
        return false
    end
    if plr.Character.Humanoid.Health <= 0 then return false end
    if plr.Character.HumanoidRootPart.Position.Y < -5 then return false end
    return true
end

local function isVisible(part, character)
    if not state.wallCheck then return true end
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, Camera, character}
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(Camera.CFrame.Position, (part.Position - Camera.CFrame.Position), rayParams)
    return result == nil
end

local function getClosestToMouse()
    local target, shortestDist = nil, state.fovRadius
    local mouseLoc = UserInputService:GetMouseLocation()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and isAlive(plr) and (not state.enemyOnly or not sameTeam(plr)) then
            local part = plr.Character:FindFirstChild(state.aimPart)
            if part then
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - mouseLoc).Magnitude
                    if dist < shortestDist and isVisible(part, plr.Character) then 
                        shortestDist = dist; target = plr 
                    end
                end
            end
        end
    end
    return target
end

task.spawn(function()
    while true do
        local customSky = Lighting:FindFirstChild("BnuuySky")
        if state.skyboxEnabled and state.isLoggedIn then
            if not customSky then
                -- Backup originals
                for _, v in pairs(Lighting:GetChildren()) do if v:IsA("Sky") then v.Parent = nil table.insert(originalSkybox, v) end end
                customSky = Instance.new("Sky", Lighting); customSky.Name = "BnuuySky"
            end
            local id = state.skyboxId
            customSky.SkyboxBk = id; customSky.SkyboxDn = id; customSky.SkyboxFt = id; customSky.SkyboxLf = id; customSky.SkyboxRt = id; customSky.SkyboxUp = id
        else
            if customSky then 
                customSky:Destroy() 
                for _, v in pairs(originalSkybox) do v.Parent = Lighting end
                originalSkybox = {}
            end
        end
        task.wait(1)
    end
end)

-- Main Rendering Loop
-- =========================
RunService.RenderStepped:Connect(function()
    if not state.active or not state.isLoggedIn then 
        fovCircle.Visible = false
        return 
    end

    if isAlive(LocalPlayer) then
        local char = LocalPlayer.Character
        local root = char.HumanoidRootPart
        local hum = char.Humanoid
        local speedActive = isFeatureActive(state.speedEnabled, state.speedKey, state.speedMode, state.speedToggled)
        hum.WalkSpeed = speedActive and state.walkSpeed or 16
        local flyActive = isFeatureActive(state.flyEnabled, state.flyKey, state.flyMode, state.flyToggled)
        if flyActive then
            hum.PlatformStand = true
            local moveDir = hum.MoveDirection
            local flyVec = Vector3.new(0,0,0)
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then flyVec = flyVec + Vector3.new(0,1,0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then flyVec = flyVec + Vector3.new(0,-1,0) end
            root.Velocity = (moveDir * state.flySpeed) + (flyVec * state.flySpeed)
        else
            hum.PlatformStand = false
        end
    end

    fovCircle.Visible = state.fovEnabled; fovCircle.Radius = state.fovRadius; fovCircle.Color = state.boneColor; fovCircle.Position = UserInputService:GetMouseLocation()

    -- AIMBOT 
    if state.aimbot and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        currentTarget = (currentTarget and isAlive(currentTarget) and (not state.enemyOnly or not sameTeam(currentTarget))) and currentTarget or getClosestToMouse()
        if currentTarget then
            local part = currentTarget.Character:FindFirstChild(state.aimPart)
            if part then Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, part.Position), 1/state.smoothing) end
        end
    else currentTarget = nil end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            if isAlive(p) then
                applyVisuals(p)
                if state.bones and (not state.enemyOnly or not sameTeam(p)) then
                    setupBones(p); local limbs = bonesData[p].limbs
                    local function connect(l, p1, p2)
                        local pt1, pt2 = p.Character:FindFirstChild(p1), p.Character:FindFirstChild(p2)
                        if pt1 and pt2 then
                            local v1, on1 = Camera:WorldToViewportPoint(pt1.Position); local v2, on2 = Camera:WorldToViewportPoint(pt2.Position)
                            if on1 and on2 then 
                                l.Visible = true; l.From = Vector2.new(v1.X, v1.Y); l.To = Vector2.new(v2.X, v2.Y); l.Color = state.boneColor; return 
                            end
                        end
                        l.Visible = false
                    end
                    connect(limbs.H_UT, "Head", "UpperTorso"); connect(limbs.UT_LT, "UpperTorso", "LowerTorso")
                    connect(limbs.UT_LUA, "UpperTorso", "LeftUpperArm"); connect(limbs.LUA_LLA, "LeftUpperArm", "LeftLowerArm")
                    connect(limbs.UT_RUA, "UpperTorso", "RightUpperArm"); connect(limbs.RUA_RLA, "RightUpperArm", "RightLowerArm")
                    connect(limbs.LT_LUL, "LowerTorso", "LeftUpperLeg"); connect(limbs.LUL_LLL, "LeftUpperLeg", "LeftLowerLeg")
                    connect(limbs.LT_RUL, "LowerTorso", "RightUpperLeg"); connect(limbs.RUL_RLL, "RightUpperLeg", "RightLowerLeg")
                elseif bonesData[p] then for _,l in pairs(bonesData[p].limbs) do l.Visible = false end end
            else
                if bonesData[p] then for _,l in pairs(bonesData[p].limbs) do l.Visible = false end end
                local char = p.Character
                if char then
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if root then
                        if root:FindFirstChild("BnuuyBox") then root.BnuuyBox.Enabled = false end
                        if root:FindFirstChild("BnuuyName") then root.BnuuyName.Enabled = false end
                        if root:FindFirstChild("BnuuyDistance") then root.BnuuyDistance.Enabled = false end
                    end
                    if char:FindFirstChild("BnuuyChams") then char.BnuuyChams.Enabled = false end
                end
            end
        end
    end
end)
